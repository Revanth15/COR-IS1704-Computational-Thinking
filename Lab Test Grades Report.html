Q4: schedule deliveries using minutes from midnight

For each sorted delivery, choose the later of the allowed earliest arrival (deadline minus 30 minutes) and the earliest arrival after the robot returns to the center. Compare against the deadline with a strict > test.

def to_minutes(clock):
    hour, minute = map(int, clock.split(":"))
    return hour * 60 + minute

def to_clock(value):
    return f"{value // 60:02d}:{value % 60:02d}"

def schedule_delivery(items):
    items = list(items)
    sort_by_latest_delivery_time(items)  # provided stable sort
    result = []
    robot_free_at_center = None

    for name, destination, latest, travel in items:
        deadline = to_minutes(latest)
        earliest_allowed = deadline - 30
        if robot_free_at_center is None:
            actual = earliest_allowed
        else:
            earliest_possible = robot_free_at_center + travel
            actual = max(earliest_allowed, earliest_possible)

        actual_text = to_clock(actual)
        if actual > deadline:
            actual_text += "*"
        result.append((name, destination, latest, actual_text))
        robot_free_at_center = actual + travel  # return trip

    return result
The initial delivery is exactly 30 minutes early. Later deliveries cannot begin until the robot has returned and traveled to the next destination. The stable provided sort preserves input order for equal deadlines.

Q4: simulate each lane against the full green/red cycle

For a car ready at time clear, find its phase in the cycle. It can start now only if it is green and its whole crossing fits before green ends. Otherwise wait to the next green. Lanes run independently, so the final count is the latest green interval any lane needs.

def calculate_actual_delays(list_cars, traffic_lights):
    green, red = traffic_lights
    cycle = green + red
    result = []
    green_count = 1

    for lane in list_cars:
        clear = 0
        lane_result = []
        for car_id, duration in lane:
            period = clear // cycle
            phase = clear % cycle
            if phase >= green or phase + duration > green:
                period += 1
                start = period * cycle
            else:
                start = clear

            clear = start + duration
            green_count = max(green_count, period + 1)
            lane_result.append((car_id, clear))
        result.append(lane_result)

    return (result, green_count)

def get_invalid_transactions(filename):

    with open(filename, "r") as file:

Q4(b): enumerate combinations and keep only the smallest sufficient sum

The transactions in this question are guaranteed valid. First reconstruct the wallet using the same exact-coin rule, then enumerate combinations. Keep every unique combination whose sum is the smallest sum at least the target.

from itertools import combinations

def get_coins(filename, address, target):
    wallets = {}
    with open(filename, "r") as file:
        for line in file:
            _, *fields = line.strip().split("|")
            inputs = []
            outputs = []
            for field in fields:
                kind, owner, raw_value = field.split(":")
                pair = (owner, int(raw_value))
                if kind == "IN":
                    inputs.append(pair)
                else:
                    outputs.append(pair)
            for owner, value in inputs:
                if owner != "COINBASE":
                    wallets[owner].remove(value)
            for owner, value in outputs:
                wallets.setdefault(owner, []).append(value)

    coins = wallets.get(address, [])
    best_sum = None
    best_combinations = set()
    for size in range(1, len(coins) + 1):
        for choice in combinations(coins, size):
            total = sum(choice)
            if total < target:
                continue
            canonical = tuple(sorted(choice))
            if best_sum is None or total < best_sum:
                best_sum = total
                best_combinations = {canonical}
            elif total == best_sum:
                best_combinations.add(canonical)

    return sorted(best_combinations)

Q4(a): two passes handle rising and falling runs

def get_min_allocation(rating):
    bonus = [1] * len(rating)
    for i in range(1, len(rating)):
        if rating[i] > rating[i - 1]:
            bonus[i] = bonus[i - 1] + 1

    for i in range(len(rating) - 2, -1, -1):
        if rating[i] > rating[i + 1]:
            bonus[i] = max(bonus[i], bonus[i + 1] + 1)
    return bonus

The first pass enforces every rising edge. The right-to-left pass enforces falling edges without undoing a larger value already needed by the first pass.

Q4(b): process cells from lower ratings to higher ratings

Strictly increasing rating edges form a directed acyclic graph. Start every cell at one point. In ascending rating order, propagate “neighbor bonus + 1” to each higher-rated neighbor. That gives the shortest valid allocation for each cell.

def get_min_allocation_grid(rating_grid):
    rows = len(rating_grid)
    cols = len(rating_grid[0])
    bonus = [[1] * cols for _ in range(rows)]
    cells = sorted(
        (rating_grid[r][c], r, c)
        for r in range(rows)
        for c in range(cols)
    )
    directions = ((0, 1), (0, -1), (1, 0), (-1, 0))

    for rating, row, col in cells:
        for dr, dc in directions:
            nr, nc = row + dr, col + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                if rating_grid[nr][nc] > rating:
                    bonus[nr][nc] = max(
                        bonus[nr][nc], bonus[row][col] + 1
                    )
    return bonus

            landing = position + jump
            if landing <= goal and map[landing] != "*":
                for suffix in routes_from(landing):
                    routes.append((jump,) + suffix)
        return tuple(routes)

    sequences = [list(route) for route in routes_from(0)]
    return (len(sequences), sequences)

Q4(b): enumerate safe right/down jumps in 2D

From each cell, try each allowed jump distance to the right and then down. Keep moves whose landing cell is safe. Since every move increases the row or column, the search cannot cycle.

from functools import lru_cache

def help_frog_against_snakes_2(map_lines):
    rows = len(map_lines)
    cols = len(map_lines[0])
    goal = (rows - 1, cols - 1)

    @lru_cache(maxsize=None)
    def routes_from(row, col):
        if (row, col) == goal:
            return ((),)
        routes = []
        for step in (1, 2, 3):
            moves = (
                (row, col + step, "r"),
                (row + step, col, "d"),
            )
            for next_row, next_col, direction in moves:
                if next_row < rows and next_col < cols:
                    if map_lines[next_row][next_col] != "*":
                        move = (step, direction)
                        for suffix in routes_from(next_row, next_col):
                            routes.append((move,) + suffix)
        return tuple(routes)

    sequences = [list(route) for route in routes_from(0, 0)]
    return (len(sequences), sequences)
